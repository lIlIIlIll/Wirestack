package wirestack_m8_006_authority_probe

import wirestack as api
import wirestack.tls as tls

func bypassHandshake(context: tls.TlsServerContext): Array<Byte> {
    match (context.externalDecryptor) {
        case Some(service) => service.decrypt(
            api.ExternalDecryptionRequest("RSA_RAW", [1u8]),
            api.OperationContext.background()
        )
        case None => []
    }
}
